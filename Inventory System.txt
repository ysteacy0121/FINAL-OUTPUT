#include <iostream>
using namespace std;

int main() {

    int choice;

    string Item_name[100];
    int Price[105], Quantity[105];
    int count = 0;
    int found;

    do {

           cout << "\n******INVENTORY MENU******\n";
        cout << "1. Add Item\n";
        cout << "2. View All Items\n";
        cout << "3. inventory report\n";
        cout << "4. delete item\n";
        cout << "5. Exit\n";
        cout << "\nENTER CHOICE: ";
        cin >> choice;

        switch (choice) {

            case 1: {
                cout << "\nItem name: ";
                cin >> Item_name[count];

                cout << "Quantity: ";
                cin >> Quantity[count];

                cout << "Price: ";
                cin >> Price[count];

                count++;

                cout << "Item added!";
                break;
            }

            case 2: {
                for (int i = 0; i < count; i++) {

                    cout << "Item name: \n" << Item_name[i];
                    cout << "Quantity: \n" << Quantity[i];
                    cout << "Price: \n" << Price[i] << endl;
                }

                if (count == 0) {
                    cout << "No items available!";
                }

                break;
            }
       
            case 3: {
                int Total_Quantity = 0;
                int Total_Value = 0;

                for (int i = 0; i < count; i++) {

                    Total_Quantity += Quantity[i];
                    Total_Value += Quantity[i] * Price[i];
                }

                cout << "\n******INVENTORY REPORT******";
                cout << "Total Items: \n" << count;
                cout << "Total Quantity: \n" << Total_Quantity;
                cout << "Total Inventory Value:\n " << Total_Value;

                break;
            }

            case 4: {
                string search;

                cout << "Enter item name to delete: ";
                cin >> search;

                found = 0;

                for (int i = 0; i < count; i++) {

                    if (Item_name[i] == search) {

                        for (int j = i; j < count - 1; j++) {

                            Item_name[j] = Item_name[j + 1];
                            Quantity[j] = Quantity[j + 1];
                            Price[j] = Price[j + 1];
                        }

                        count--;

                        cout << "Item deleted!";

                        found = 1;
                        break;
                    }
                }

                if (found == 0) {
                    cout << "Item not found!";
                }

                break;
            }

            case 5: {
                cout << "Exiting Inventory System...";
                break;
            }

            default: {
                cout << "Invalid choice!";
                break;
            }

        }

    } while (choice != 5);

    return 0;
}
